#include <stdio.h>
#include <unistd.h>
#include <sys/types.h>

int main() {
    pid_t pid;

    printf("Before forking, current PID: %d\n", getpid());

    // Create a child process
    pid = fork();

    if (pid < 0) {
        // Scenario 1: Fork failed
        fprintf(stderr, "Fork Failed\n");
        return 1;
    } 
    else if (pid == 0) {
        // Scenario 2: Child Process
        // fork() returns 0 to the child
        printf("\n--- Child Process Block ---\n");
        printf("I am the Child!\n");
        printf("My PID: %d\n", getpid());
        printf("My Parent's PID (PPID): %d\n", getppid());
    } 
    else {
        // Scenario 3: Parent Process
        // fork() returns the Child's PID to the parent
        printf("\n--- Parent Process Block ---\n");
        printf("I am the Parent!\n");
        printf("My PID: %d\n", getpid());
        printf("My Child's PID: %d\n", pid);
        
        // Wait briefly to let the child finish its output
        sleep(1); 
    }

    return 0;
}
